# the other loaders' lists are read from the class path by their bare name: a root
# server-unsupported-tests.txt would shadow theirs in the nested layers.

buildcraft/tests/wooden_pipe_boxes-Test.grug
coverage/tests/block_shape-Test.grug

# The GUI a server opens for its player. Test.screen_open is forwarded to the joined client now

# GUI.open until #1098 gives it a server-side container, so the menu never opens there.
# #1094's forwarded_screen_open and forwarded_close_screen run here instead.
coverage/tests/remote_gui-Test.grug

# The screen input #1094 forwards. A forwarded test presses a key or turns the wheel on the
# client's screen, and this server cannot open one for its player until #1098, so these run only
# on 1.20.6.
coverage/tests/forwarded_press_screen_key-Test.grug
coverage/tests/forwarded_scroll_screen-Test.grug
